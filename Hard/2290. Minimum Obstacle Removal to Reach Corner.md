# 2290. Minimum Obstacle Removal to Reach Corner
Problem : [HERE](https://leetcode.com/problems/minimum-obstacle-removal-to-reach-corner/description/)



```CPP
class Solution {
public:
    int minimumObstacles(vector<vector<int>>& grid) {        
        //二維陣列，先row再column
        int row = grid.size(), column = grid[0].size();

        //定義方向，上下左右
        vector<pair<int, int>> direction={{1, 0}, {-1, 0}, {0, 1}, {0, -1}};

        //最短距離陣列
        vector<vector<int>> dist(row, vector<int>(column, INT_MAX));

        //雙端隊列
        deque<pair<int, int>> dq;

        //起點距離
        dist[0][0] = 0;
        //起點加入deque
        dq.push_front({0, 0});

        //開始走
        while(!dq.empty()){
            //當下位置
            auto [x, y] = dq.front();
            //移除當下位置
            dq.pop_front();

            //往不同方向移動
            for(auto [dx, dy] : direction){
                //移動過後的位置
                int nx = x + dx , ny = y + dy;

                //確保沒有超過邊界
                if(nx >= 0 && nx < row && ny >= 0 && ny < column){
                    //計算新的距離
                    int newDist = dist[x][y] + grid[nx][ny];

                    //找到更短距離
                    if(newDist < dist[nx][ny]){
                        dist[nx][ny] = newDist;
                        if(grid[nx][ny] == 0) dq.push_front({nx, ny});
                        else dq.push_back({nx,ny});
                    }
                }
            }
        }
        return dist[row - 1][column - 1];
    }
};
```
