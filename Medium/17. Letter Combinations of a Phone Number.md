# 17. Letter Combinations of a Phone Number
Problem : [HERE](https://leetcode.com/problems/letter-combinations-of-a-phone-number/description/)

>[!TIP]
>用遞迴的方式，Current Digit Index到Digit Len - 1的時候就把答案存起來
>
>前面要先用一個暫存的 遞迴才能進行

```C
/**
 * Note: The returned array must be malloced, assume caller calls free().
 */
void combine(char** ans , char** map , char* previousTempAns , char* digits , int currentDigitIndex , int digitlen , int* anscount)
{    
    int len = strlen(map[digits[currentDigitIndex] - '0']);
    printf("Len in Digit : %d\nCurrent Digit Index : %d\n" , len , currentDigitIndex);
    for(int i = 0 ; i < len ; i++){
        char tempans[currentDigitIndex + 2];
        strcpy(tempans , previousTempAns);
        tempans[currentDigitIndex] = map[digits[currentDigitIndex] - '0'][i];
        tempans[currentDigitIndex + 1] = '\0';
        printf("Previous Temp Ans : %s\nTemp Ans : %s\nAns count : %d\n" , previousTempAns , tempans , *anscount);
        if(currentDigitIndex == digitlen - 1){
            strcpy(ans[(*anscount)++] , tempans);
        }
        else{
            combine(ans , map , tempans , digits , currentDigitIndex + 1 , digitlen , anscount);
        }
    }
}

char** letterCombinations(char* digits, int* returnSize) {
    char* map[10] = {" " , " " , "abc" , "def" , "ghi" , "jkl" , "mno" , "pqrs" , "tuv" , "wxyz"};
    int len = strlen(digits);
    printf("Digits Len : %d\n" , len);
    if(len == 0) {
       *returnSize = 0;
       return NULL;
    }
    int count = 0;
    for(int i = 0 ; i < len ; i++){
        if(digits[i] == '7' || digits[i] == '9') count++;
    }
    int size = (int)pow(3 , len - count) * (int)pow(4 , count);
    printf("Size : %d\n" , size);
    *returnSize = size;
    char** ans = malloc(size * sizeof(char*));
    for(int i = 0 ; i < size ; i++){
        ans[i] = malloc((len + 1) * sizeof(char));
    }

    int anscount = 0;
    char tempans[len + 1];
    tempans[0] = '\0';
    combine(ans , map , tempans , digits ,  0 , len , &anscount);

    return ans;
}
```
