# 670. Maximum Swap
Problem : [HERE](https://leetcode.com/problems/maximum-swap/description/)




```CPP
class Solution {
public:
    int maximumSwap(int num) {
        string numString = to_string(num);
        // 初始化 vector，長度是10、初始值是-1
        vector<int>numPosition(10 ,-1);
        
        // 記錄每個數字最後出現的位置
        for(int i = 0 ; i < numString.size() ; i++){
            numPosition[numString[i] - '0'] = i;
        }

        // 從左到右找到一個數字，它的右邊有一個更大的數字
        for(int i = 0 ; i < numString.size() ; i++){
            // 從最左邊的數字開始
            for(int j = 9 ; j > numString[i] - '0' ; j--){
                // 從最大的數字9 開始往下找 看有沒有比當前數字更大的
                //有比現在的數字大的話，看他的位置有沒有比現在還後面
                if(numPosition[j] > i){
                    // 交換 numStr[i] 和 numStr[last[d]]
                    swap(numString[i] , numString[numPosition[j]]);
                    // 轉換回數字並返回
                    return stoi(numString);
                }
            }
        }
        // 如果無法進行交換，返回原始數字
        return num;
    }
};
```
