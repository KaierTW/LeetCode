# 3479. Fruits Into Baskets III
Problem : [HERE](https://leetcode.com/problems/fruits-into-baskets-iii/description/)

>[!TIP]
>
>## 🧠 一、題目解題思路與邏輯統整
>
>### 題目概要：
>你有一組 `fruits[]` 表示水果重量，以及一組 `baskets[]` 表示每個籃子的最大承重容量。每個籃子只能用一次，你要將水果分配到能承重的籃子中，求**無法被放進任何籃子的水果數量**。
>
>---
>
>### 核心解題流程：
>
>#### ✅ Step 1：排序籃子
>
>```cpp
>vector<pair<int, int>> sortedBaskets; // {capacity, originalIndex}
>for (int i = 0; i < n; ++i)
>    sortedBaskets.emplace_back(baskets[i], i);
>sort(sortedBaskets.begin(), sortedBaskets.end());
>```
>
>* **目的**：為了快速找出第一個「承重 ≥ 該水果」的籃子。
>* **原始 index 保留**：因為後面我們要知道原本的順序誰最小。
>
>---
>
>#### ✅ Step 2：建立 Segment Tree
>
>* 範圍為 `0 ~ n-1`（對應 `sortedBaskets` 中的 index）
>* 每個節點的值是「該區間中原始 index 最小的值」。
>
>```cpp
>vector<int> originalIndices(n);
>for (int i = 0; i < n; ++i)
>    originalIndices[i] = sortedBaskets[i].second;
>seg.build(originalIndices);
>```
>
>---
>
>#### ✅ Step 3：對每顆水果執行：
>
>1. **Binary Search 找第一個能放這顆水果的籃子 index（在 sortedBaskets 中）**
>
>   * 找 `capacity >= fruit` 的第一個位置 `idx`
>
>2. **從 Segment Tree 查詢區間 `idx ~ n-1`**：
>
>   * 找出區間中「原始 index 最小」的可用籃子（代表該重量下可用的**最左邊**籃子）
>
>3. **若查無籃子 → 無法放置 → `++unplaced`**
>
>   * 否則：
>
>     * 找出那個原始 index 對應在 `sortedBaskets` 中的位置 → 更新為 `INF`，代表此籃子已使用。
>
>---
>
>### ❓常見疑問整理
>
>#### Q1：為什麼 Binary Search 後還要再找 Segment Tree？
>
>* Binary Search 只能找「哪裡開始有足夠容量」。
>* 但這裡面可能有多個籃子滿足條件 → 我們要找「原始 index 最小」者（靠左者），避免之後可用籃子更少。
>
>#### Q2：為什麼要保留原始 index？
>
>* 我們希望使用最靠左邊（原始排列）可用籃子 → 最小化對未來水果的影響。
>
>#### Q3：為什麼要用 Segment Tree？
>
>* 每次找「idx \~ n-1 中原始 index 最小的值」，如果用 linear search 會太慢（O(n)），Segment Tree 可加速成 O(log n)。
>
>---




```CPP
class SegmentTree{
    int n;
    vector<int> tree;
public:
    SegmentTree(int size){
        n = size;
        tree.assign(4 * n, INT_MAX);
    }
    
    void build(const vector<int>& data, int node, int left, int right){
        //只剩一個元素就可以記錄了
        if(left == right){
            tree[node] = data[left];            
        }
        else{
            int mid = (left + right) / 2;
            // 建立左子樹
            build(data, node * 2, left, mid);
            // 建立右子樹
            build(data, node * 2 + 1, mid + 1, right);
            tree[node] = min(tree[node * 2], tree[node * 2 + 1]);
        }
    }

    void build(const vector<int>& data) {
        build(data, 1, 0, n - 1);
    }

    int query(int node, int left, int right, int queryLeft, int queryRight){
        // 完全不重疊：跳過
        if (queryRight < left || right < queryLeft) return INT_MAX;

        // 完全包含：直接回傳
        if (queryLeft <= left && right <= queryRight) return tree[node];

        // 部分重疊：分左右遞迴查詢
        int mid = (left + right) / 2;
        int leftMin = query(node * 2, left, mid, queryLeft, queryRight);
        int rightMin = query(node * 2 + 1, mid + 1, right, queryLeft, queryRight);

        return min(leftMin, rightMin);
    }
    int query(int left, int right){
        return query(1, 0, n - 1, left, right);
    }

    void update(int node, int left, int right, int index, int value) {
        if (left == right) {
            tree[node] = value;
        } else {
            int mid = (left + right) / 2;
            if (index <= mid)
                update(node * 2, left, mid, index, value);
            else
                update(node * 2 + 1, mid + 1, right, index, value);
            tree[node] = min(tree[node * 2], tree[node * 2 + 1]);
        }
    }

    void update(int idx, int value) {
        update(1, 0, n - 1, idx, value);
    }
};

class Solution {
public:
    int numOfUnplacedFruits(vector<int>& fruits, vector<int>& baskets) {
        // Hint 1: Sort the baskets by the pair of (basket[i], i) in the array.
        vector<pair<int, int>> sortedBaskets;// {capacity, original index}
        int n = baskets.size();
        for(int i = 0 ; i < n ; i++){
            sortedBaskets.push_back({baskets[i], i});
        }
        // sortedBaskets依照capacity從小排到大
        sort(sortedBaskets.begin(), sortedBaskets.end());
        // 把排序後的「原始 index」拿出來，建立一個陣列
        vector<int> originalIndex(n);
        unordered_map<int, int> indexInSorted; // 新增：原始 index -> 排序後的 index
        for(int i = 0 ; i < n ; i++){
            originalIndex[i] = sortedBaskets[i].second;
            indexInSorted[sortedBaskets[i].second] = i;
        }
        SegmentTree segTree(n);
        segTree.build(originalIndex);        

        int ans = 0;
        // Hint 2: For each fruit from left to right, use binary search to find the first index in the sorted array such that basket[i] >= fruit.
        for(int fruit : fruits){
            int left = 0, right = n - 1, index = n;
            while(left <= right){
                int mid = (left + right) / 2;
                if(sortedBaskets[mid].first >= fruit){
                    // 找看看左邊還有沒有也是能裝的
                    index = mid;                    
                    right = mid - 1;
                }
                else{
                    left = mid + 1;
                }                
            }
            // 沒有能裝的
            if(index == n){
                ans++;
                continue;
            }

            // 找原先baskets最左邊能裝的下的Index
            int bestIndex = segTree.query(index, n - 1);
            if(bestIndex == INT_MAX){
                ans++;                
            }
            else{
                // 找到最左邊的Index要把他標記成用過的
                int sortedIdx = indexInSorted[bestIndex];
                segTree.update(sortedIdx, INT_MAX);
            }        
        }
        return ans;
    }
};
```
