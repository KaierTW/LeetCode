# 22. Generate Parentheses
Problem : [HERE](https://leetcode.com/problems/generate-parentheses/description/)

>[!TIP]
>用遞迴，一個一個找進去，當長度=2 * n的時候就傳進答案
>
>先從'('開始，紀錄個數，'('的數量要先比')'多

```C
/**
 * Note: The returned array must be malloced, assume caller calls free().
 */
void addtoAns(int n , char** ans , int* returnSize , char* temp , int tempIndex , int left , int right)
{
    if(tempIndex == 2 * n){
        ans[*returnSize] = malloc((1 + 2 * n) * sizeof(char));
        strcpy(ans[(*returnSize)++] , temp);
        return;
    }
    if(left < n){
        temp[tempIndex] = '(';
        addtoAns(n , ans , returnSize , temp , tempIndex + 1 , left + 1 , right);
    }
    if(right < left){
        temp[tempIndex] = ')';
        addtoAns(n , ans , returnSize , temp , tempIndex + 1 , left , right + 1);
    }

}

char** generateParenthesis(int n, int* returnSize) {
    char** ans = malloc(pow(n ,n) * sizeof(char*));
    *returnSize = 0;
    char* temp = malloc((1 + 2 * n) * sizeof(char));
    temp[2 * n] = '\0';
    int tempIndex = 0;
    int left = 0;
    int right = 0;
    addtoAns(n , ans , returnSize , temp , tempIndex , left , right);
    free(temp);
    return ans;
}
```
