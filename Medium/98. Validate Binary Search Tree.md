# 98. Validate Binary Search Tree
Problem : [HERE](https://leetcode.com/problems/validate-binary-search-tree/description/)

>[!TIP]
>要按照

```C
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     struct TreeNode *left;
 *     struct TreeNode *right;
 * };
 */

void Valid(struct TreeNode* root , int* arr , int* index){
    if(root == NULL) return ;
    Valid(root -> left , arr , index);
    arr[(*index)++] = root -> val;
    Valid(root -> right , arr , index);
}

bool isValidBST(struct TreeNode* root) {
    int* arr = malloc (100000 * sizeof(int));
    int index = 0;
    Valid(root , arr , &index);
    for(int i = 0 ; i < index - 1 ; i++){
        if(arr[i] >= arr[i+1]) return false;
    }
    return true;
}
```
