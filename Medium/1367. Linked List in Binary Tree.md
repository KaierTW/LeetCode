# 1367. Linked List in Binary Tree
Problem : [HERE](https://leetcode.com/problems/linked-list-in-binary-tree/description/)

>[!TIP]
>要先看當前node有沒有符合第一個List
>
>有的話才用函數遞迴去找
>
>沒有的話就往下找

```CPP
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:
    bool isSubPath(ListNode* head, TreeNode* root) {
        if(root == NULL) return false;
        if(head -> val == root -> val){
            if(findIsSubPath(head , root))
            return true;
        }
        return isSubPath(head , root -> left) || isSubPath(head , root -> right);
    }
    bool findIsSubPath(ListNode* head, TreeNode* root){
        if(root == NULL) return false;
        cout << "Now head val : " << head -> val << " , Now root val : " << root -> val << endl;
        if(head -> val != root -> val) return false;
        if(head -> next  == NULL) return true;                        
        return findIsSubPath(head -> next , root -> left) || findIsSubPath(head -> next , root -> right);
    }
};
```
