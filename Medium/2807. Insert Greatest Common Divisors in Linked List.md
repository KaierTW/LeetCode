# 2807. Insert Greatest Common Divisors in Linked List
Problem : [HERE](https://leetcode.com/problems/insert-greatest-common-divisors-in-linked-list/description/)

>[!TIP]
>每兩個原本的node中間都要加gcd，所以下一個head應該是gcd的next，也就是原來head的next

```CPP
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    ListNode* insertGreatestCommonDivisors(ListNode* head) {
        ListNode* ans = head;
        while(head -> next != NULL){
            int a = head -> val;
            int b = head -> next -> val;
            int gcdValue = gcd(a ,b);
            ListNode* gcdNode = new ListNode(gcdValue , head -> next);
            head -> next = gcdNode;
            head = gcdNode -> next;
        }
        return ans;
    }
};
```
