# 1717. Maximum Score From Removing Substrings
Problem : [HERE](https://leetcode.com/problems/maximum-score-from-removing-substrings/description)

>[!TIP]
>先找哪一個分數比較高，要先刪除分數比較多的字
>
>用stack的方式，先存到一個暫存的，當有存到分數比較多的開頭就可以來比較看看是否符合
>
>都先判定完分數比較多的，再拿剛剛暫存的去處理分數比較少的字

```C
int maximumGain(char* s, int x, int y) {    
    int ans = 0;
    char big[3] , small[3];
    int bigScore , smallScore;
    if(x > y){
        strcpy(big , "ab");
        strcpy(small , "ba");
        bigScore = x;
        smallScore = y;
    }
    else{
        strcpy(big , "ba");
        strcpy(small , "ab");
        bigScore = y;
        smallScore = x;
    }

    char bigTemp[strlen(s) + 1];
    int bigTempIndex = 0;
    for(int i = 0 ; s[i] != '\0' ; i++){
        if(bigTempIndex > 0 && bigTemp[bigTempIndex - 1] == big[0] && s[i] == big[1]){
            ans += bigScore;
            bigTempIndex--;
        }
        else{
            bigTemp[bigTempIndex++] = s[i];
        }
    }

    char smallTemp[strlen(s) + 1];
    int smallTempIndex = 0;
    for(int i = 0 ; i < bigTempIndex ; i++){
        if(smallTempIndex > 0 && smallTemp[smallTempIndex - 1] == small[0] && bigTemp[i] == small[1]){
            ans += smallScore;
            smallTempIndex--;
        }
        else{
            smallTemp[smallTempIndex++] = bigTemp[i];
        }
    }
    return ans;
}
```
