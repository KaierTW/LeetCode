# 113. Path Sum II
Problem : [HERE](https://leetcode.com/problems/path-sum-ii/description/)

>[!TIP]
>可以參考[112. Path Sum](../Easy/112.%20Path%20Sum.md)
>
>只是多加入了把走過的路徑記錄起來


```CPP
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:
    void findPath(TreeNode* node , int target , int sum , vector<int>& path , vector<vector<int>>& ans){
        if(node == NULL) return;
        path.push_back(node -> val);
        sum += node -> val;
        if(sum == target && node -> left == NULL && node -> right == NULL){
            ans.push_back(path);
        }
        else{
            findPath(node -> left , target , sum , path , ans);
            findPath(node -> right , target , sum , path , ans);
        }
        path.pop_back();
    }

    vector<vector<int>> pathSum(TreeNode* root, int targetSum) {
        vector<vector<int>> ans;
        vector<int> path;
        int sum = 0;
        findPath(root , targetSum , sum , path , ans);
        return ans;
    }
};
```
