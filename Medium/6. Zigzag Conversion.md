# 6. Zigzag Conversion
Problem : [HERE](https://leetcode.com/problems/zigzag-conversion/description/)


```C
char* convert(char* s, int numRows) {
    int len = strlen(s);
    char* ans = malloc((len + 1) * sizeof(char));
    int index = 0;
    for(int i = 0 ; i < numRows ; i++){
        int temp = i;
        if(i == 0 || i == numRows - 1){
            while(temp < len){                
                ans[index++] = s[temp];
                if(numRows == 1) temp++;                
                else temp += 2 * numRows - 2;
            }
        }
        else{
            int flag = 0;
            while(temp < len){
                if(flag == 0){
                    ans[index++] = s[temp];
                    temp += 2 * numRows - 2 - 2 * i;
                    flag = 1;
                }
                else{
                    ans[index++] = s[temp];
                    temp += 2 * i;
                    flag = 0; 
                }

            }
        }
    }
    ans[len] = '\0';
    return ans;
}
```
