# 8. String to Integer (atoi)
Problem : [HERE](https://leetcode.com/problems/string-to-integer-atoi/description/)

>[!TIP]
>1.遇到空白的先看有沒有遇到數字過，有的話→回傳數字，沒有的話→繼續找
>
>2.判斷正負號
>
>3.處理數字，看有沒有超過INT_MAX，有的話→正的回傳INT_MAX；負的回傳INT_MIN，沒有的話→繼續找

```C
int myAtoi(char* s) {
    int len = strlen(s);
    long int ans = 0 , flag = 0 , isVal = 0;
    if(len == 0) return 0;
    for(int i = 0 ; i < len ; i++){
        printf("%d\n",ans);
        if(s[i] == ' ') {
            if(isVal) return (flag) ? -ans : ans;
            else continue;
        }
        else if(!isVal && (s[i] == '+' || s[i] == '-')){
            if(s[i + 1] <= '9' && s[i + 1] >= '0') flag = (s[i] == '-') ? 1 : 0;
            else return 0;
        }
        else if (s[i] <= '9' && s[i] >= '0'){            
            isVal = 1;
            ans = ans * 10 + s[i] - '0';
            if(ans > INT_MAX) return flag ? INT_MIN : INT_MAX;
        }
        else break;        
    }
    return (flag) ? -ans : ans;
}
```
