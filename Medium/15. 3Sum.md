/**
 * Return an array of arrays of size *returnSize.
 * The sizes of the arrays are returned as *returnColumnSizes array.
 * Note: Both returned array and *columnSizes array must be malloced, assume caller calls free().
 */
 void BubbleSort(int* nums , int size){
    for(int i = 0 ; i < size - 1 ; i++){
        for(int j = 0 ; j < size - i - 1 ; j++){
            if(nums[j] > nums[j + 1]){
                int temp = nums[j];
                nums[j] = nums[j + 1];
                nums[j + 1] = temp;
            }
        }
    }
 }
int** threeSum(int* nums, int numsSize, int* returnSize, int** returnColumnSizes) {
    *returnSize = 0;
    int** ans = malloc((numsSize * numsSize) * sizeof(int*));
    //int** ans = malloc(sizeof(int*));
    //*returnColumnSizes = malloc((numsSize * numsSize) * sizeof(int*));
    *returnColumnSizes = malloc((numsSize * numsSize) * sizeof(int));    
    //*returnColumnSizes = malloc(sizeof(int));
    BubbleSort(nums , numsSize);
    for(int i = 0 ; i < numsSize - 2 ; i++){
        if(i != 0 && nums[i] == nums[i - 1]) continue;
        int second = i + 1;
        int third = numsSize - 1;
        while(second < third){
            int sum = nums[i] + nums[second] + nums[third];
            if(sum == 0){
                ans[*returnSize] = malloc(3 * sizeof(int));
                ans[*returnSize][0] = nums[i];
                ans[*returnSize][1] = nums[second];
                ans[*returnSize][2] = nums[third];
                (*returnColumnSizes)[*returnSize] = 3;                          
                (*returnSize)++;
                //ans = realloc(ans , ((*returnSize + 1) * sizeof(int*)));
                //printf("Pointer : %p\n" , ans);
                //*returnColumnSizes = realloc(*returnColumnSizes , ((*returnSize + 1) * sizeof(int)));                
                while(second < third && nums[second] == nums[second + 1]) second++;
                while(second < third && nums[third] == nums[third - 1]) third--;
                second++;
                third--;
            }
            else if(sum < 0) second++;
            else if(sum > 0) third--;
        }
    }
    return ans;
}
