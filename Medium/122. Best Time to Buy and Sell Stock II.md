>
>### 方法一：貪婪算法
>這種方法通過在每次價格上升時進行買賣操作來最大化利潤：
>
>```cpp
>#include <iostream>
>#include <vector>
>
>class Solution {
>public:
>    int maxProfit(std::vector<int>& prices) {
>        int maxProfit = 0;
>        for (size_t i = 1; i < prices.size(); ++i) {
>            if (prices[i] > prices[i - 1]) {
>                maxProfit += prices[i] - prices[i - 1];
>            }
>        }
>        return maxProfit;
>    }
>};
>
>int main() {
>    std::vector<int> prices = {7, 1, 5, 3, 6, 4};
>    Solution solution;
>    std::cout << "Max Profit: " << solution.maxProfit(prices) << std::endl;
>    return 0;
>}
>// 輸出：Max Profit: 7
>```
>
>### 方法二：動態規劃
>這種方法使用動態規劃來計算最大利潤：
>
>```cpp
>#include <iostream>
>#include <vector>
>#include <algorithm>  // std::max
>
>class Solution {
>public:
>    int maxProfit(std::vector<int>& prices) {
>        int sell = 0, hold = INT_MIN;
>        for (const int price : prices) {
>            sell = std::max(sell, hold + price);
>            hold = std::max(hold, sell - price);
>        }
>        return sell;
>    }
>};
>
>int main() {
>    std::vector<int> prices = {7, 1, 5, 3, 6, 4};
>    Solution solution;
>    std::cout << "Max Profit: " << solution.maxProfit(prices) << std::endl;
>    return 0;
>}
>// 輸出：Max Profit: 7
>```

```CPP
class Solution {
public:
    int maxProfit(vector<int>& prices) {
        int profit = 0;
        for(int i = 1 ; i < prices.size() ; i++){
            if(prices[i] > prices[i - 1]) profit += (prices[i] - prices[i - 1]);
        }
        return profit;
    }
};
```
