# 874. Walking Robot Simulation
Problem : [HERE](https://leetcode.com/problems/walking-robot-simulation/description/)

>[!TIP]
>可以用set集合的方式來檢測障礙物
>
>set 是一個集合，用來存儲不重複的唯一元素，且這些元素是有序的。set 通常基於紅黑樹（平衡二叉搜索樹）實現，因此元素按照鍵值進行排序。每個元素只能出現一次，並且可以快速進行插入、刪除和查找操作。
>
>pair 是一個用來存儲兩個相關聯數據的模板類。它可以用來存儲兩個不同類型或相同類型的變數。


```CPP
class Solution {
public:
    int robotSim(vector<int>& commands, vector<vector<int>>& obstacles) {
        vector<int> ans(2,0); //兩個元素，起始值都是0
        int direction = 0; //0:北 1:東 2:南 3:西
        int maxDistance = 0;

        //用集合來檢測障礙物
        set<pair<int , int>> obstaclesSet;
        for(const auto& obstacle : obstacles){
            obstaclesSet.insert({obstacle[0] , obstacle[1]});
        }

        //不同方位的移動量
        vector<pair<int , int>> move = {{0 , 1} , {1 , 0} , {0 , -1} , {-1 , 0}}; //北、東、南、西

        for(const int& command : commands){
            if(command == -2){
                //左轉90度
                direction = (direction + 3) % 4;
            }
            else if(command == -1){
                //右轉90度
                direction = (direction + 1) % 4;
            }
            else{
                //往前走
                for(int i = 0 ; i < command ; i++){
                    int newX = ans[0] + move[direction].first;
                    int newY = ans[1] + move[direction].second;

                    //檢查下一步會不會撞到障礙物
                    if(obstaclesSet.count({newX , newY})){
                        break;
                    }

                    //沒有撞到障礙物，更新ans
                    ans[0] = newX;
                    ans[1] = newY;

                    //更新max
                    maxDistance= max(maxDistance , ans[0] * ans[0] + ans[1] * ans[1]);
                }
            }
        }
        return maxDistance;
    }
};
```
