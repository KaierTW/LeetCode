# 31. Next Permutation
Problem : [HERE](https://leetcode.com/problems/next-permutation/description/)

>[!TIP]
>要找到下一個排列，從整個數列的後面往前找
>
>如果當前數字(i)比他前面的數字(i - 1)還要大，就要換位子
>
>換位子的時候也要從後面開始往前找，找到比i - 1還要大的時候，就把兩個換位子
>
>然後再重新排列

```CPP
class Solution {
public:    
    int findNext(vector<int>& nums , int i){
        for(int j = nums.size() - 1 ; j > i - 1 ; j--){
            if(nums[j] > nums[i - 1]) return j;
        }
        return i;
    }
    void nextPermutation(vector<int>& nums) {
        int index;
        for(int i = nums.size() - 1 ; i > 0 ; i--){
            if(nums[i] > nums[i - 1]){
                index = findNext(nums , i);
                swap(nums[i - 1] , nums[index]);
                sort(nums.begin() + i , nums.end());
                return;
            }
        }
        sort(nums.begin() , nums.end());
    }
};
```
