# 12. Integer to Roman
Problem : [HERE](https://leetcode.com/problems/integer-to-roman/description/)

>[!TIP]
>注意可以用減的4 (IV), 9 (IX), 40 (XL), 90 (XC), 400 (CD) and 900 (CM).

```C
char* intToRoman(int num) {    
    char* ans = malloc(50 * sizeof(int));
    int count = 0;
    while(num > 0){
        if(num >= 1000){
            *(ans + count++) = 'M';
            num -= 1000;
        }
        else if(num >= 900){
            *(ans + count++) = 'C';
            *(ans + count++) = 'M';
            num -= 900;
        }
        else if(num >= 500){
            *(ans + count++) = 'D';
            num -= 500;
        }
        else if(num >= 400){
            *(ans + count++) = 'C';
            *(ans + count++) = 'D';
            num -= 400;
        }
        else if(num >= 100){
            *(ans + count++) = 'C';
            num -= 100;
        }
        else if(num >= 90){
            *(ans + count++) = 'X';
            *(ans + count++) = 'C';
            num -= 90;
        }
        else if(num >= 50){
            *(ans + count++) = 'L';
            num -= 50;
        }
        else if(num >= 40){
            *(ans + count++) = 'X';
            *(ans + count++) = 'L';
            num -= 40;
        }
        else if(num >= 10){
            *(ans + count++) = 'X';
            num -= 10;
        }
        else if(num >= 9){
            *(ans + count++) = 'I';
            *(ans + count++) = 'X';
            num -= 9;
        }
        else if(num >= 5){
            *(ans + count++) = 'V';
            num -= 5;
        }
        else if(num >= 4){
            *(ans + count++) = 'I';
            *(ans + count++) = 'V';
            num -= 4;
        }
        else if(num >= 1){
            *(ans + count++) = 'I';
            num -= 1;
        }        
    }
    *(ans + count) = '\0';
    return ans;
}
```
