# 73. Set Matrix Zeroes
Problem : [HERE](https://leetcode.com/problems/set-matrix-zeroes/description/)

>[!TIP]
>這題 **Set Matrix Zeroes** 的核心思路是：如果矩陣中的某個元素是 `0`，則將該元素所在的整行和整列都設為 `0`。但關鍵是要 **原地修改**（in-place），不能使用額外的矩陣來存儲狀態。
>
>### 解法：
>1. **使用額外空間（O(m + n)）**
>   - 建立兩個陣列 `rowFlags` 和 `colFlags` 來記錄哪些行和列需要被設為 `0`。
>   - 先遍歷矩陣，記錄所有 `0` 所在的行和列。
>   - 再次遍歷矩陣，根據 `rowFlags` 和 `colFlags` 設置對應的元素為 `0`。
>
>2. **最佳解法（O(1) 空間）**
>   - **利用矩陣的第一行和第一列作為標記**：
>     - 先檢查第一行和第一列是否本身有 `0`，用兩個變數 `rowZero` 和 `colZero` 記錄。
>     - 遍歷矩陣（不包括第一行和第一列），如果某個元素是 `0`，則將該元素所在的行的第一個元素設為 `0`，列的第一個元素設為 `0`。
>     - 再次遍歷矩陣（不包括第一行和第一列），根據第一行和第一列的標記來設置對應的元素為 `0`。
>     - 最後根據 `rowZero` 和 `colZero` 設置第一行和第一列。
>
>這樣就能在 **O(1) 額外空間** 下完成矩陣修改。




## **解法1**
```CPP
class Solution {
public:
    void setZeroes(vector<vector<int>>& matrix) {
        int row = matrix.size(), column = matrix[0].size();
        set<int> setRow;
        set<int> setColumn;
        for(int r = 0 ; r < row ; r++){
            for(int c  = 0 ; c < column ; c++){
                if(matrix[r][c] == 0) {
                    setRow.insert(r);
                    setColumn.insert(c);
                }
            }
        }
        for(int r = 0 ; r < row ; r++){
            if(setRow.find(r) != setRow.end()){
                for(int c = 0 ; c < column ; c++){
                    matrix[r][c] = 0;
                }
            }
        }
        for(int c = 0 ; c < column ; c++){
            if(setColumn.find(c) != setColumn.end()){
                for(int r = 0 ; r < row ; r++){
                    matrix[r][c] = 0;
                }
            }
        }
    }
};
```

## **解法2**
```CPP
#include <vector>
using namespace std;

class Solution {
public:
    void setZeroes(vector<vector<int>>& matrix) {
        int rows = matrix.size(), cols = matrix[0].size();
        bool rowZero = false, colZero = false;

        // 檢查第一行和第一列是否需要變為 0
        for (int i = 0; i < rows; i++) {
            if (matrix[i][0] == 0) colZero = true;
        }
        for (int j = 0; j < cols; j++) {
            if (matrix[0][j] == 0) rowZero = true;
        }

        // 使用第一行和第一列作為標記
        for (int i = 1; i < rows; i++) {
            for (int j = 1; j < cols; j++) {
                if (matrix[i][j] == 0) {
                    matrix[i][0] = 0;
                    matrix[0][j] = 0;
                }
            }
        }

        // 根據標記設置矩陣元素為 0
        for (int i = 1; i < rows; i++) {
            if (matrix[i][0] == 0) {
                for (int j = 1; j < cols; j++) {
                    matrix[i][j] = 0;
                }
            }
        }
        for (int j = 1; j < cols; j++) {
            if (matrix[0][j] == 0) {
                for (int i = 1; i < rows; i++) {
                    matrix[i][j] = 0;
                }
            }
        }

        // 處理第一行和第一列
        if (rowZero) {
            for (int j = 0; j < cols; j++) matrix[0][j] = 0;
        }
        if (colZero) {
            for (int i = 0; i < rows; i++) matrix[i][0] = 0;
        }
    }
};

```
