# 2326. Spiral Matrix IV
Problem : [HERE](https://leetcode.com/problems/spiral-matrix-iv/description/)

>[!TIP]
>先建立一個二維陣列，然後都初始化為-1
>
>接著記錄邊界，然後從最上面的row開始，順時針走，經過最右邊column、最下面row、最左邊column，慢慢繞圈就可以

```CPP
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    vector<vector<int>> spiralMatrix(int m, int n, ListNode* head) {
        vector<vector<int>> ans(m , vector<int>(n , -1));
        int topRow = 0 , bottomRow = m - 1 , leftColumn = 0 , rightColumn = n - 1;
        while(head){
            //Fill top row
            for(int i = leftColumn ; i <= rightColumn && head ; i++){
                ans[topRow][i] = head -> val;
                head = head -> next;
            }
            topRow++;

            //Fill right column
            for(int i = topRow ; i <= bottomRow && head ; i++){                
                ans[i][rightColumn] = head -> val;
                head = head -> next;
            }
            rightColumn--;

            //Fill bottom row
            for(int i = rightColumn ; i >= leftColumn && head ; i--){
                ans[bottomRow][i] = head -> val;
                head = head -> next;
            }
            bottomRow--;

            //Fill left column
            for(int i = bottomRow ; i >= topRow && head ; i--){
                ans[i][leftColumn] = head -> val;
                head = head -> next;
            }
            leftColumn++;
        }
        return ans;
    }
};
```
