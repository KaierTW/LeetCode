>
>### 方法一：使用三次反轉
>這種方法通過三次反轉來實現旋轉：
>
>1. 反轉整個數組。
>2. 反轉前`k`個元素。
>3. 反轉剩下的元素。
>
>```cpp
>#include <iostream>
>#include <vector>
>#include <algorithm>  // std::reverse
>
>class Solution {
>public:
>    void rotate(std::vector<int>& nums, int k) {
>        int n = nums.size();
>        k %= n;  // 確保k不超過數組長度
>        std::reverse(nums.begin(), nums.end());  // 反轉整個數組
>        std::reverse(nums.begin(), nums.begin() + k);  // 反轉前k個元素
>        std::reverse(nums.begin() + k, nums.end());  // 反轉剩下的元素
>    }
>};
>
>int main() {
>    std::vector<int> nums = {1, 2, 3, 4, 5, 6, 7};
>    int k = 3;
>    Solution().rotate(nums, k);
>    for (int num : nums) {
>        std::cout << num << " ";
>    }
>    return 0;
>}
>// 輸出：5 6 7 1 2 3 4
>```
>
>### 方法二：使用額外空間
>這種方法使用一個臨時數組來存儲旋轉後的結果：
>
>```cpp
>#include <iostream>
>#include <vector>
>
>class Solution {
>public:
>    void rotate(std::vector<int>& nums, int k) {
>        int n = nums.size();
>        std::vector<int> temp(n);
>        for (int i = 0; i < n; ++i) {
>            temp[(i + k) % n] = nums[i];
>        }
>        nums = temp;
>    }
>};
>
>int main() {
>    std::vector<int> nums = {1, 2, 3, 4, 5, 6, 7};
>    int k = 3;
>    Solution().rotate(nums, k);
>    for (int num : nums) {
>        std::cout << num << " ";
>    }
>    return 0;
>}
>// 輸出：5 6 7 1 2 3 4
>```

```CPP
class Solution {
public:
    void rotate(vector<int>& nums, int k) {
        k %= nums.size();
        // 反轉整個數組
        reverse(nums.begin(), nums.end());
        // 反轉前k個元素
        reverse(nums.begin(), nums.begin() + k);
        // 反轉剩下的元素
        reverse(nums.begin() + k, nums.end());
    }
};
```
