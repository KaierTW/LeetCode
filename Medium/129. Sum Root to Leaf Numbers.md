# 129. Sum Root to Leaf Numbers
Problem : [HERE]()

>[!TIP]
>
>## 🧠 解題思路
>
>這題的核心是使用 **DFS（深度優先搜尋）** 來遍歷整棵樹，並在每條從根到葉的路徑上累加形成的數字。
>
>### 步驟如下：
>
>1. **從根節點開始**，一路往下走。
>2. 每走到一個節點，就把目前的數字乘以 10 再加上該節點的值（模擬數字拼接）。
>3. 如果走到葉節點（左右子節點都為空），就把目前的數字加到總和中。
>4. 最後回傳所有路徑形成的數字總和。
>
>---
>
>## 💻 C++ 程式碼範例
>
>```cpp
>class Solution {
>public:
>    int sumNumbers(TreeNode* root) {
>        return dfs(root, 0);
>    }
>
>private:
>    int dfs(TreeNode* node, int currentSum) {
>        if (!node) return 0;
>
>        currentSum = currentSum * 10 + node->val;
>
>        // 如果是葉節點，回傳目前累積的數字
>        if (!node->left && !node->right) {
>            return currentSum;
>        }
>
>        // 遞迴左右子樹
>        return dfs(node->left, currentSum) + dfs(node->right, currentSum);
>    }
>};
>```
>
>---
>

```CPP
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:
    void dfs(TreeNode* node, string &temp, int &sum){
        if(node == NULL) return;
        temp += to_string(node -> val);
        if(node -> left == NULL && node -> right == NULL){
            sum += stoi(temp);
            cout << "Sum: " << sum << " Temp: " << temp << endl;
            temp.pop_back();
            return;
        }
        dfs(node -> left, temp, sum);        
        dfs(node -> right, temp, sum);
        temp.pop_back();
    }
    int sumNumbers(TreeNode* root) {
        string temp;
        int sum = 0;
        dfs(root, temp, sum);
        return sum;
    }
};
```
