# 16. 3Sum Closest
Problem : [HERE](https://leetcode.com/problems/3sum-closest/description/)

>[!TIP]
>跟 [15.3Sum](./15.%203Sum.md) 一樣的概念
>
>從第陣列一個數字開始，然後設定左邊跟右邊，如果sum > target，要減少右邊元素，所以要往左邊移動
>
>如果sum < target，要增加左邊的元素，所以要往右邊移動


```C
 void BubbleSort(int* nums , int size){
    for(int i = 0 ; i < size - 1 ; i++){
        for(int j = 0 ; j < size - i - 1 ; j++){
            if(nums[j] > nums[j + 1]){
                int temp = nums[j];
                nums[j] = nums[j + 1];
                nums[j + 1] = temp;
            }
        }
    }
 }

int threeSumClosest(int* nums, int numsSize, int target) {
    BubbleSort(nums , numsSize);
    printf("New nums : [ ");
    for(int i =0 ; i < numsSize ; i++){
        printf("%d ",nums[i]);
    }
    printf("]\n");
    int minDiff = INT_MAX;
    int ans = 0;
    for(int i = 0 ; i < numsSize - 2 ; i++){
        int left = i + 1;
        int right = numsSize - 1;
        while(left < right){
            int sum = nums[i] + nums[left] + nums[right];
            //printf("Sum : %d\n",sum);
            if(sum == target) return target;
            else if(sum > target){    
                if(sum - target < minDiff){
                    minDiff = sum - target;
                    ans = sum;
                }
                right--;
            }
            else if(sum < target){
                if(target - sum < minDiff){
                    minDiff = target - sum;
                    ans = sum;
                }
                left++;
            }            
            //printf("Min Diff : %d" , minDiff);
        }
    }
    return ans;
}
```
