# 592. Fraction Addition and Subtraction
Problem : [HERE](https://leetcode.com/problems/fraction-addition-and-subtraction/description/)



```CPP
class Solution {
public:
    string fractionAddition(string expression) {
        int i = 0 , size = expression.size() , ansNum = 0 , ansDen = 1;

        while(i < size){
            int flag = 1;
            if(expression[i] == '+' || expression[i] == '-'){
                if(expression[i] == '-') flag = -1;                
                i++;
            }

            int num = 0;
            while(i < size && isdigit(expression[i])){
                num = num * 10 + (expression[i++] - '0');
            }
            num *= flag;

            i++;

            int den = 0;
            while(i < size && isdigit(expression[i])){
                den = den * 10 + (expression[i++] - '0');
            }

            ansNum = ansNum * den + num * ansDen;
            ansDen *= den;

            int gcdVal = gcd(abs(ansNum) , ansDen);
            ansNum /= gcdVal;
            ansDen /= gcdVal;
                    
            cout << "ansNum: " << ansNum << " ansDen: " << ansDen << " GCD: " << gcdVal <<endl;        
        }
        return to_string(ansNum) + "/" + to_string(ansDen);
    }
};
```
