# 2501. Longest Square Streak in an Array
Problem : [HERE](https://leetcode.com/problems/longest-square-streak-in-an-array/description/)


```CPP
class Solution {
public:
    int longestSquareStreak(vector<int>& nums) {
 // 排序數組
    sort(nums.begin(), nums.end());
    
    // 使用集合加速查找
    unordered_set<int> num_set(nums.begin(), nums.end());
    
    int maxLength = 0;
    int limit = sqrt(INT_MAX);  // 計算不會超出 int 的最大平方根

    // 遍歷每個數字
    for (int num : nums) {
        int length = 0;
        
        // 嘗試從 num 開始構造平方序列
        while (num_set.count(num)) {
            length++;
            // 檢查是否會發生溢出
            if (num > limit) break;
            num = num * num;  // 計算平方
        }
        
        maxLength = max(maxLength, length);  // 更新最長長度
    }

    // 如果沒有找到符合條件的平方序列，回傳 -1
    return maxLength > 1 ? maxLength : -1;
    }
};
```
