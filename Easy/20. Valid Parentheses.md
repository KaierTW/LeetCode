# 20. Valid Parentheses
Problem : [HERE](https://leetcode.com/problems/valid-parentheses/description/)

```C
bool isValid(char* s) {
    int len = strlen(s);
    if(len < 2) return false;
    char* stack = malloc(len * sizeof(char));
    int stackIndex = 0;
    for(int i = 0 ; i < len ; i++){
        if (s[i] == '('){
            stack[stackIndex++] = '(';
        }
        else if (s[i] == '['){
            stack[stackIndex++] = '[';
        }
        else if (s[i] == '{'){
            stack[stackIndex++] = '{';
        }
        else if (s[i]==')'){
            if(i == 0 || stackIndex == 0) return false;
            else if(stack[--stackIndex] == '(') continue;
            else return false;
        }
        else if (s[i]==']'){
            if(i == 0 || stackIndex == 0) return false;
            else if(stack[--stackIndex] == '[') continue;
            else return false;
        }
        else if (s[i]=='}'){
            if(i == 0 || stackIndex == 0) return false;
            else if(stack[--stackIndex] == '{') continue;
            else return false;
        }
    }
    if(stackIndex == 0) return true;
    else return false;
    
}
```
