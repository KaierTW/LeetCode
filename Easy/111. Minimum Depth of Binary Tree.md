# 111. Minimum Depth of Binary Tree
Problem : [HERE](https://leetcode.com/problems/minimum-depth-of-binary-tree/description/)

>[!TIP]
>基本上跟 [104. Maximum Depth of Binary Tree](104.%20Maximum%20Depth%20of%20Binary%20Tree)一樣
>
>只是改成回傳比較小的深度
>
>要注意如果root如果有一邊是沒有東西的話，是要回傳另一邊的深度

```CPP
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:
    int findDepth(TreeNode* node){
        if(node == NULL) return 0;
        else if(node -> left == NULL) return findDepth(node -> right) + 1;
        else if(node -> right == NULL) return findDepth(node -> left) + 1;
        else{
            int leftDepth = findDepth(node -> left) + 1;
            int rightDepth = findDepth(node -> right) + 1;
            if(leftDepth < rightDepth) return leftDepth;
            else return rightDepth;
        }
    }
    int minDepth(TreeNode* root) {
        return findDepth(root);
    }
};
```
